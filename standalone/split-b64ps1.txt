#!/usr/bin/env pwsh
# split-b64.ps1 - Split large base64 file into email-sized chunks

param(
    [Parameter(Mandatory=$true)]
    [string]$Base64File,
    [int]$ChunkSize = 50000  # 50K chars per chunk
)

if (-not (Test-Path $Base64File)) {
    Write-Host "File not found: $Base64File" -ForegroundColor Red
    exit 1
}

$content = [System.IO.File]::ReadAllText($Base64File)
$totalLength = $content.Length
$chunkCount = [Math]::Ceiling($totalLength / $ChunkSize)

Write-Host "Splitting $totalLength chars into $chunkCount chunks of ~$ChunkSize chars each" -ForegroundColor Cyan

for ($i = 0; $i -lt $chunkCount; $i++) {
    $start = $i * $ChunkSize
    $length = [Math]::Min($ChunkSize, $totalLength - $start)
    $chunk = $content.Substring($start, $length)
    
    $chunkFile = "$Base64File.part$($i + 1)"
    [System.IO.File]::WriteAllText($chunkFile, $chunk)
    Write-Host "Created: $chunkFile ($length chars)" -ForegroundColor Green
}
